import 'package:flutter/material.dart';
import 'package:flutter/rendering.dart';

void main() {
  runApp(FlutterApp());
}

class FlutterApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: "Demo App",
      debugShowCheckedModeBanner: false,
      theme: ThemeData(primarySwatch: Colors.blue),
      home: DemoScreen(),
    );
  }
}

class DemoScreen extends StatelessWidget {
  @override
  Widget build(Object context) {
    return Scaffold(
      appBar: AppBar(title: Text("Demo Screen")),

      body: Column(
        children: [ItmBar(), ListBarItems(), ItmsShow(), BottomItems()],
      ),
    );
  }
}

class ItmBar extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Expanded(
      flex: 1,
      child: Container(
        color: Colors.red,
        width: double.infinity,
        child: ListView.builder(
          scrollDirection: Axis.horizontal,

          itemBuilder: (context, index) {
            return Padding(
              padding: const EdgeInsets.all(8.0),
              child: CircleAvatar(radius: 40, child: Text('${index + 0}')),
            );
          },
          itemCount: 10,
        ),
      ),
    );
  }
}

class ListBarItems extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    var arrNames = [
      "Alice",
      "Bob",
      "Charlie",
      "Diana",
      "Ethan",
      "Fiona",
      "George",
      "Hannah",
      "Ian",
      "Jane",
      "Kevin",
      "Laura",
      "Mike",
      "Nina",
      "Oscar",
      "Paula",
      "Quinn",
      "Rachel",
      "Steve",
      "Tina",
    ];
    return Expanded(
      flex: 4,
      child: Container(
        color: Colors.blue,
        width: double.infinity,
        child: ListView.builder(
          itemBuilder: (context, index) {
            return Card(
              elevation: 10,
              child: ListTile(
                leading: CircleAvatar(
                  backgroundImage: AssetImage('assets/images/Flutter.webp'),
                  radius: 25,
                ),
                title: Text("${arrNames[index]}"),
                subtitle: Text("Subtitle for ${arrNames[index]}"),
                trailing: Icon(Icons.arrow_forward_ios),
              ),
            );
          },
          itemCount: arrNames.length,
        ),
      ),
    );
  }
}

class ItmsShow extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Expanded(
      flex: 2,
      child: Container(
        color: Colors.grey,
        width: double.infinity,
        child: ListView.builder(
          scrollDirection: Axis.horizontal,
          itemBuilder: (context, index) {
            return Container(
              margin: EdgeInsets.all(10),
              color: Colors.white,
              height: 40,
              width: 200,
              child: Center(child: Text("Item ${index + 1}")),
            );
          },
          itemCount: 5,
        ),
      ),
    );
  }
}

class BottomItems extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return Expanded(
      flex: 1,
      child: Container(
        color: Colors.green,
        width: double.infinity,
        child: ListView.builder(
          scrollDirection: Axis.horizontal,
          itemBuilder: (context, index) {
            return CircleAvatar(radius: 60, child: Text("A"));
          },
          itemCount: 5,
        ),
      ),
    );
  }
}
